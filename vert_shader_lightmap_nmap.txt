
uniform mat4 mWorldView;
uniform mat4 mProj;
uniform mat4 invWorld;

varying vec2 texcoord;
varying vec2 texcoord2;
varying vec3 normal_in;
varying vec3 incident_light_in;

void main(void)
{
	gl_Position = mProj * mWorldView * gl_Vertex;

	//gl_TexCoord[0] = gl_MultiTexCoord0;
	//gl_TexCoord[1] = gl_MultiTexCoord1;

	incident_light_in = gl_Color.xyz;
	normal_in = vec4(invWorld * vec4(gl_Normal,0)).xyz;

	texcoord = vec2(gl_MultiTexCoord0);
	texcoord2 = vec2(gl_MultiTexCoord1);
}
