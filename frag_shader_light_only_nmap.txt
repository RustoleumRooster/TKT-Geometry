
uniform sampler2D myTexture;
uniform sampler2D myTexture2;
uniform sampler2D myTexture3;

varying vec2 texcoord;
varying vec2 texcoord2;
varying vec3 normal_in;
varying vec3 incident_light_in;

void main (void)
{
	
    	vec4 raw_nmap = texture2D(myTexture3, texcoord);
	vec4 col = texture2D(myTexture, texcoord);
	vec3 nmap;

	nmap.b = (raw_nmap.b* 2.0) - 1.0;
	nmap.r = (raw_nmap.r* 2.0) - 1.0;
	nmap.g = (raw_nmap.g* 2.0) - 1.0;

	vec3 incident_light;
	incident_light.x = (incident_light_in.x* 2.0) - 1.0;
	incident_light.y = (incident_light_in.y* 2.0) - 1.0;
	incident_light.z = (incident_light_in.z* 2.0) - 1.0;

	float light = 0.2 + 0.8* texture2D(myTexture2, texcoord2).r;

	clamp(light,0,1);

	vec3 normal = normalize(normal_in);
	vec3 R = normalize(vec3(1,-1,0));

	vec3 Tan = cross(normal,vec3(0,1,0));

	if(normal == vec3(0,1,0))
		Tan = vec3(1,0,0);
	else if(normal == vec3(0,-1,0))
		Tan = vec3(-1,0,0);

	vec3 BiTan = cross(Tan,normal);

	//vec3 N = normal * col.b + BiTan * col.r + Tan * col.g;
	vec3 N = normal * nmap.b + Tan * nmap.r + BiTan * nmap.g;

	float p = dot(incident_light,N);

	gl_FragColor = col * p * light;
	//gl_FragColor = vec4(1,1,1,1) * p;
}
